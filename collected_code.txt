def require_roles(*allowed_roles: str):
    """
    Restrict an endpoint to specified database-backed user roles.
    """

    def role_checker(
        current_user: Dict[str, Any] = Depends(get_current_user),
    ):
        if current_user.get("role") not in allowed_roles:
            raise HTTPException(
                status_code=403,
                detail="You do not have permission to perform this action.",
            )

        return current_user

    return role_checker

    conn = get_conn()

        user = conn.execute(
            """

        event_type="email_verification_success",
        actor_user_id=user["id"],
        target_email=email,
    )

    return {

def verify_login_otp(
    payload: OTPVerifyRequest,
    response: Response,
):
    """Verify login OTP and establish an authenticated session."""

    email = payload.email.strip().lower()

    # Retrieve the account associated with the email.
    from .database import get_conn

    conn = get_conn()

    try:
        user = conn.execute(
            """
            SELECT id, email, role, is_active, email_verified_utc
            FROM auth_users
            WHERE email = ?
            """,
            (email,)
        ).fetchone()
    finally:
        conn.close()

    if (
        not user
        or not user["is_active"]
        or not user["email_verified_utc"]
    ):
        record_auth_event(
            event_type="login_otp_failed",
            target_email=email,
        )
        raise HTTPException(
            status_code=401,
            detail="Invalid or expired verification code."
        )

    # Verify and consume the OTP.
    valid = verify_otp(
        user_id=user["id"],
        code=payload.code,
        purpose="login",
    )

    if not valid:
        record_auth_event(
            event_type="login_otp_failed",
            actor_user_id=user["id"],
            target_email=email,
        )
        raise HTTPException(
            status_code=401,
            detail="Invalid or expired verification code."
        )

    # Create a server-managed session.
    try:
        session = create_session(user["id"])
    except ValueError:
        raise HTTPException(
            status_code=403,
            detail="Unable to establish an authenticated session."
        )

    # Store the session token in an HTTP-only cookie.
    response.set_cookie(
        key=AUTH_COOKIE_NAME,
        value=session["token"],
        max_age=AUTH_SESSION_HOURS * 3600,
        httponly=True,
        secure=AUTH_COOKIE_SECURE,
        samesite="lax",
        path="/",
    )

    record_auth_event(
        event_type="login_success",
        actor_user_id=user["id"],
        target_email=email,
    )

    return {
        "success": True,
        "authenticated": True,
        "user": session["user"],
        "expires_utc": session["expires_utc"],
    }

OTP_RESEND_COOLDOWN_SECONDS = 60

VALID_ROLES = {"super_admin", "admin", "viewer"}


# --------------------------------------------------
# Shared utilities

def utc_now() -> datetime:
    """Return the current timezone-aware UTC datetime."""
    return datetime.now(timezone.utc)


def utc_string(value: Optional[datetime] = None) -> str:
    """Convert a datetime to a consistent UTC ISO timestamp."""
    value = value or utc_now()
    return value.astimezone(timezone.utc).isoformat()


def hash_token(token: <REDACTED>
    """Create a keyed digest for sensitive authentication tokens."""
    if not AUTH_SECRET_KEY:
        raise RuntimeError("AUTH_SECRET_KEY is not configured.")

    return hmac.new(
        AUTH_SECRET_KEY.encode("utf-8"),
        token.encode("utf-8"),
        hashlib.sha256
    ).hexdigest()


def normalize_email(email: str) -> str:
    """Normalize an email address."""
    return email.strip().lower()


def validate_email(email: str) -> bool:
    """Perform basic email address validation."""
    email = normalize_email(email)

    return bool(
        re.fullmatch(
            r"[A-Za-z0-9.!#$%&'*+/=?^_`{|}~-]+@"
            r"[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}"
            r"[A-Za-z0-9])?"
            r"(?:\.[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}"
            r"[A-Za-z0-9])?)+",
            email
        )
    )

def authenticate_password(email: str, password: <REDACTED>
    """
    Verify user credentials without exposing whether an email exists.

    Returns the account details when credentials are valid,
    otherwise returns None.
    """

    if not email or not password:
        return None

    email = normalize_email(email)

    if not validate_email(email):
        return None

    conn = get_conn()

    try:
        user = conn.execute(
            """
            SELECT
                id,
                email,
                password_hash,
                role,
                is_active,
                email_verified_utc
            FROM auth_users
            WHERE email = ?
            """,
            (email,)
        ).fetchone()

        if not user:
            return None

        if not user["is_active"]:
            return None

        if not verify_password(password, user["password_hash"]):
            return None

        return {
            "id": user["id"],
            "email": user["email"],
            "role": user["role"],
            "email_verified": bool(user["email_verified_utc"])
        }

    finally:
        conn.close()

def issue_otp(user_id: int, purpose: str = "login") -> str:
    """
    Generate and store a six-digit OTP.

    Returns the plaintext code internally so the email service
    can deliver it. Never expose it through a public API response.
    """

    if purpose not in {"login", "email_verification"}:
        raise ValueError("Unsupported OTP purpose.")

    now = utc_now()
    now_text = utc_string(now)
    expiry_text = utc_string(
        now + timedelta(minutes=OTP_EXPIRY_MINUTES)
    )

    code = f"{secrets.randbelow(1_000_000):06d}"
    code_hash = hash_token(f"otp:{user_id}:{purpose}:{code}")

    conn = get_conn()

    try:
        conn.execute("BEGIN IMMEDIATE")

        user = conn.execute(
            """
            SELECT id, is_active
            FROM auth_users
            WHERE id = ?
            """,
            (user_id,)
        ).fetchone()

        if not user or not user["is_active"]:
            raise ValueError("User does not exist or is inactive.")

        latest = conn.execute(
            """
            SELECT created_utc
            FROM auth_otps
            WHERE user_id = ? AND purpose = ?
            ORDER BY created_utc DESC
            LIMIT 1
            """,
            (user_id, purpose)
        ).fetchone()

        if latest:
            latest_time = datetime.fromisoformat(
                latest["created_utc"]
            )

            elapsed = (now - latest_time).total_seconds()

            if elapsed < OTP_RESEND_COOLDOWN_SECONDS:
                raise ValueError(
                    "Please wait before requesting another OTP."
                )

        # Invalidate previously issued, unused OTPs.
        conn.execute(
            """
            UPDATE auth_otps
            SET invalidated_utc = ?
            WHERE user_id = ?
              AND purpose = ?
              AND consumed_utc IS NULL
              AND invalidated_utc IS NULL
            """,
            (now_text, user_id, purpose)
        )

        conn.execute(
            """
            INSERT INTO auth_otps (
                user_id,
                otp_hash,
                purpose,
                attempts,
                max_attempts,
                created_utc,
                expires_utc
            )
            VALUES (?, ?, ?, 0, ?, ?, ?)
            """,
            (
                user_id,
                code_hash,
                purpose,
                OTP_MAX_ATTEMPTS,
                now_text,
                expiry_text
            )
        )

        conn.commit()
        return code

    except Exception:
        conn.rollback()
        raise

    finally:
        conn.close()


def verify_otp(
    user_id: int,
    code: str,
    purpose: str = "login"
) -> bool:
    """
    Verify the latest valid OTP.

    A successful OTP is consumed immediately.
    Failed attempts are recorded in the database.
    """

    if purpose not in {"login", "email_verification"}:
        return False

    if not re.fullmatch(r"\d{6}", code or ""):
        return False

    now = utc_now()
    now_text = utc_string(now)

    conn = get_conn()

    try:
        conn.execute("BEGIN IMMEDIATE")

        otp = conn.execute(
            """
            SELECT *
            FROM auth_otps
            WHERE user_id = ?
              AND purpose = ?
              AND consumed_utc IS NULL
              AND invalidated_utc IS NULL
            ORDER BY created_utc DESC
            LIMIT 1
            """,
            (user_id, purpose)
        ).fetchone()

        if not otp:
            conn.commit()
            return False

        expiry = datetime.fromisoformat(otp["expires_utc"])

        if now >= expiry:
            conn.execute(
                """
                UPDATE auth_otps
                SET invalidated_utc = ?
                WHERE id = ?
                """,
                (now_text, otp["id"])
            )
            conn.commit()
            return False

        if otp["attempts"] >= otp["max_attempts"]:
            conn.execute(
                """
                UPDATE auth_otps
                SET invalidated_utc = ?
                WHERE id = ?
                """,
                (now_text, otp["id"])
            )
            conn.commit()
            return False

        submitted_hash = hash_token(
            f"otp:{user_id}:{purpose}:{code}"
        )

        if not hmac.compare_digest(
            submitted_hash,
            otp["otp_hash"]
        ):
            new_attempts = otp["attempts"] + 1

            conn.execute(
                """
                UPDATE auth_otps
                SET attempts = ?,
                    invalidated_utc = CASE
                        WHEN ? >= max_attempts THEN ?
                        ELSE invalidated_utc
                    END
                WHERE id = ?
                """,
                (
                    new_attempts,
                    new_attempts,
                    now_text,
                    otp["id"]
                )
            )

            conn.commit()
            return False

        conn.execute(
            """
            UPDATE auth_otps
            SET consumed_utc = ?
            WHERE id = ?
            """,
            (now_text, otp["id"])
        )

        conn.commit()
        return True

    except Exception:
        conn.rollback()
        raise

    finally:
        conn.close()

def get_session_user(raw_token: <REDACTED>
    """
    Validate a session token and return its authenticated user.

    Returns None for missing, expired, revoked, or invalid sessions.
    """

    if not raw_token:
        return None

    token_digest = hash_token(f"session:{raw_token}")
    now = utc_now()
    now_text = utc_string(now)

    conn = get_conn()

    try:
        conn.execute("BEGIN IMMEDIATE")

        row = conn.execute(
            """
            SELECT
                s.id AS session_id,
                s.expires_utc,
                s.last_seen_utc,
                u.id AS user_id,
                u.email,
                u.full_name,
                u.role,
                u.is_active,
                u.email_verified_utc
            FROM auth_sessions s
            JOIN auth_users u ON u.id = s.user_id
            WHERE s.session_token_hash = ?
              AND s.revoked_utc IS NULL
            """,
            (token_digest,)
        ).fetchone()

        if not row:
            conn.commit()
            return None

        expires = datetime.fromisoformat(row["expires_utc"])

        if (
            now >= expires
            or not row["is_active"]
            or not row["email_verified_utc"]
        ):
            conn.commit()
            return None

        # Update last_seen at most once every five minutes.
        last_seen = (
            datetime.fromisoformat(row["last_seen_utc"])
            if row["last_seen_utc"]
            else None
        )

        if (
            last_seen is None
            or (now - last_seen).total_seconds() >= 300
        ):
            conn.execute(
                """
                UPDATE auth_sessions
                SET last_seen_utc = ?
                WHERE id = ?
                """,
                (now_text, row["session_id"])
            )

        conn.commit()

        return {
    "id": row["user_id"],
    "email": row["email"],
    "full_name": row["full_name"],
    "role": row["role"],
    "session_id": row["session_id"]
}

    except Exception:
        conn.rollback()
        raise

    finally:
        conn.close()

def create_invitation(
    email: str,
    role: str,
    invited_by: int,
    validity_hours: int = 48
) -> dict:
    """Create a secure, time-limited account invitation."""

    email = normalize_email(email)

    if not validate_email(email):
        raise ValueError("Invalid email address.")

    if role not in {"admin", "viewer"}:
        raise ValueError("Invalid invitation role.")

    if not 1 <= validity_hours <= 168:
        raise ValueError("Invitation validity must be 1-168 hours.")

    raw_token = <REDACTED>
    token_digest = hash_token(f"invite:{raw_token}")

    now = utc_now()
    expires = now + timedelta(hours=validity_hours)

    conn = get_conn()

    try:
        conn.execute("BEGIN IMMEDIATE")

        actor = conn.execute(
            """
            SELECT role, is_active
            FROM auth_users
            WHERE id = ?
            """,
            (invited_by,)
        ).fetchone()

        if not actor or not actor["is_active"]:
            raise ValueError("Inviting administrator is invalid.")

        if actor["role"] != "super_admin":
           raise ValueError("Only the super administrator can invite users.")

        existing_user = conn.execute(
            """
            SELECT id
            FROM auth_users
            WHERE email = ?
            """,
            (email,)
        ).fetchone()

        if existing_user:
            raise ValueError("An account already exists for this email.")

        # Revoke any previous outstanding invitations.
        conn.execute(
            """
            UPDATE auth_invitations
            SET revoked_utc = ?
            WHERE email = ?
              AND accepted_utc IS NULL
              AND revoked_utc IS NULL
            """,
            (utc_string(now), email)
        )

        conn.execute(
            """
            INSERT INTO auth_invitations (
                email,
                token_hash,
                role,
                invited_by,
                created_utc,
                expires_utc
            )
            VALUES (?, ?, ?, ?, ?, ?)
            """,
            (
                email,
                token_digest,
                role,
                invited_by,
                utc_string(now),
                utc_string(expires)
            )
        )

        conn.execute(
            """
            INSERT INTO auth_audit_log (
                actor_user_id,
                event_type,
                target_email,
                details_json,
                created_utc
            )
            VALUES (?, ?, ?, ?, ?)
            """,
            (
                invited_by,
                "invitation_created",
                email,
                '{"role": "' + role + '"}',
                utc_string(now)
            )
        )

        conn.commit()

        return {
            "email": email,
            "role": role,
            "token": raw_token,
            "expires_utc": utc_string(expires)
        }

    except Exception:
        conn.rollback()
        raise

    finally:
        conn.close()



def accept_invitation(
    raw_token: <REDACTED>
    full_name: str,
    password: <REDACTED>
) -> dict:
    """Redeem an invitation and create an unverified user account."""

    if not raw_token:
        raise ValueError("Invitation token is required.")

    full_name = full_name.strip()

    if not full_name:
        raise ValueError("Full name is required.")

    if len(full_name) > 120:
        raise ValueError("Full name must not exceed 120 characters.")

    password_digest = hash_password(password)
    token_digest = hash_token(f"invite:{raw_token}")

    now = utc_now()
    now_text = utc_string(now)

    conn = get_conn()

    try:
        conn.execute("BEGIN IMMEDIATE")

        invitation = conn.execute(
            """
            SELECT *
            FROM auth_invitations
            WHERE token_hash = ?
              AND accepted_utc IS NULL
              AND revoked_utc IS NULL
            """,
            (token_digest,)
        ).fetchone()

        if not invitation:
            raise ValueError("Invitation is invalid or already used.")

        expires = datetime.fromisoformat(
            invitation["expires_utc"]
        )

        if now >= expires:
            raise ValueError("Invitation has expired.")

        existing_user = conn.execute(
            """
            SELECT id
            FROM auth_users
            WHERE email = ?
            """,
            (invitation["email"],)
        ).fetchone()

        if existing_user:
            raise ValueError("An account already exists for this email.")

        cursor = conn.execute(
            """
            INSERT INTO auth_users (
    email,
    full_name,
    password_hash,
    role,
    is_active,
    email_verified_utc,
    created_utc,
    updated_utc
)
VALUES (?, ?, ?, ?, 1, NULL, ?, ?)
            """,
            (
    invitation["email"],
    full_name,
    password_digest,
    invitation["role"],
    now_text,
    now_text
)
        )

        user_id = cursor.lastrowid

        conn.execute(
            """
            UPDATE auth_invitations
            SET accepted_utc = ?
            WHERE id = ?
            """,
            (now_text, invitation["id"])
        )

        conn.execute(
            """
            INSERT INTO auth_audit_log (
                actor_user_id,
                event_type,
                target_email,
                details_json,
                created_utc
            )
            VALUES (?, ?, ?, ?, ?)
            """,
            (
                invitation["invited_by"],
                "invitation_accepted",
                invitation["email"],
                '{"new_user_id": ' + str(user_id) + '}',
                now_text
            )
        )

        conn.commit()

        return {
            "user_id": user_id,
            "email": invitation["email"],
            "full_name": full_name, 
            "role": invitation["role"],
            "email_verified": False
        }

    except Exception:
        conn.rollback()
        raise

    finally:
        conn.close()



def verify_email_otp(user_id: int, code: str) -> bool:
    """
    Verify an email OTP and activate email verification.

    OTP consumption and account verification happen
    in one atomic database transaction.
    """

    purpose = "email_verification"

    if not re.fullmatch(r"\d{6}", code or ""):
        return False

    now = utc_now()
    now_text = utc_string(now)

    conn = get_conn()

    try:
        conn.execute("BEGIN IMMEDIATE")

        # Find the latest unused email verification OTP.
        otp = conn.execute(
            """
            SELECT *
            FROM auth_otps
            WHERE user_id = ?
              AND purpose = ?
              AND consumed_utc IS NULL
              AND invalidated_utc IS NULL
            ORDER BY created_utc DESC
            LIMIT 1
            """,
            (user_id, purpose)
        ).fetchone()

        if not otp:
            conn.commit()
            return False

        # Check expiration.
        expiry = datetime.fromisoformat(otp["expires_utc"])

        if now >= expiry:
            conn.execute(
                """
                UPDATE auth_otps
                SET invalidated_utc = ?
                WHERE id = ?
                """,
                (now_text, otp["id"])
            )
            conn.commit()
            return False

        # Check maximum attempts.
        if otp["attempts"] >= otp["max_attempts"]:
            conn.execute(
                """
                UPDATE auth_otps
                SET invalidated_utc = ?
                WHERE id = ?
                """,
                (now_text, otp["id"])
            )
            conn.commit()
            return False

        # Compare the submitted code against its stored hash.
        submitted_hash = hash_token(
            f"otp:{user_id}:{purpose}:{code}"
        )

        if not hmac.compare_digest(
            submitted_hash,
            otp["otp_hash"]
        ):
            new_attempts = otp["attempts"] + 1

            conn.execute(
                """
                UPDATE auth_otps
                SET attempts = ?,
                    invalidated_utc = CASE
                        WHEN ? >= max_attempts THEN ?
                        ELSE invalidated_utc
                    END
                WHERE id = ?
                """,
                (
                    new_attempts,
                    new_attempts,
                    now_text,
                    otp["id"]
                )
            )

            conn.commit()
            return False

        # Mark the account as email verified.
        cursor = conn.execute(
            """
            UPDATE auth_users
            SET email_verified_utc = ?,
                updated_utc = ?
            WHERE id = ?
              AND is_active = 1
              AND email_verified_utc IS NULL
            """,
            (now_text, now_text, user_id)
        )

        # Do not consume the OTP if the account cannot be verified.
        if cursor.rowcount != 1:
            conn.rollback()
            return False

        # Consume the OTP in the same transaction.
        conn.execute(
            """
            UPDATE auth_otps
            SET consumed_utc = ?
            WHERE id = ?
            """,
            (now_text, otp["id"])
        )

        conn.commit()
        return True

    except Exception:
        conn.rollback()
        raise

    finally:
        conn.close()

            raise ValueError("LSTM V8 API returned an invalid response")

import asyncio
import logging
from datetime import datetime, timezone
from typing import Any, Dict, Optional

from .lstm_v8_client import LSTMV8Client

logger = logging.getLogger(__name__)


class LSTMV8Service:
    """Polls LSTM V8 independently and caches its latest successful state."""

    def __init__(self, poll_interval_sec: float = 5.0) -> None:
        self.client = LSTMV8Client()
        self.poll_interval_sec = max(2.0, poll_interval_sec)

        self.latest_state: Optional[Dict[str, Any]] = None
        self.last_success_utc: Optional[str] = None
        self.last_error: Optional[str] = None

        self.connected = False
        self._task: Optional[asyncio.Task] = None
        self._stop_event = asyncio.Event()

    def get_status(self) -> Dict[str, Any]:
        return {
            "connected": self.connected,
            "last_success_utc": self.last_success_utc,
            "last_error": self.last_error,
            "state": self.latest_state,
        }

    async def refresh_once(self) -> None:
        try:
            state = await asyncio.to_thread(
                self.client.get_dashboard_state
            )

            self.latest_state = state
            self.last_success_utc = datetime.now(
                timezone.utc
            ).isoformat()

            self.last_error = None
            self.connected = True

        except Exception as exc:
            self.connected = False
            self.last_error = str(exc)

            logger.warning(
                "LSTM V8 state refresh failed: %s",
                exc,
            )

    async def _poll_loop(self) -> None:
        while not self._stop_event.is_set():
            await self.refresh_once()

            try:
                await asyncio.wait_for(
                    self._stop_event.wait(),
                    timeout=self.poll_interval_sec,
                )
            except asyncio.TimeoutError:
                pass

    async def start(self) -> None:
        if self._task and not self._task.done():
            return

        self._stop_event.clear()
        self._task = asyncio.create_task(
            self._poll_loop()
        )

    async def stop(self) -> None:
        self._stop_event.set()

        if self._task:
            await self._task
            self._task = None

class PipelineOrchestrator:
    def __init__(self) -> None:
        self.connector = ClusterTelemetryConnector()
        self.features = RollingFeaturePipeline(maxlen=96)
        self.ml_engine = ModelEngine()
        self.lstm_v8 = LSTMV8Service(poll_interval_sec=5.0)
        self.alerts = SmartAlertEngine()

        self.chart_history: Dict[str, Deque[Dict[str, Any]]] = {}
        self.cluster_history: Deque[Dict[str, Any]] = deque(maxlen=72)
        self.latest_snapshot: Dict[str, Any] = {}
        self.clients: Set[WebSocket] = set()

        self.stream_interval_sec: float = 3.0
        self.paused: bool = False
        self._tick_counter: int = 0

    def sync_interval_from_settings(self) -> None:
        try:
            s = get_app_settings()
            interval = float(s.get("refresh_interval_sec", 3.0))
            self.stream_interval_sec = max(1.0, min(60.0, interval))
        except Exception:
            logger.exception("Could not synchronize dashboard refresh interval.")

    def step_once(self) -> Dict[str, Any]:
        self._tick_counter += 1

        batch = self.connector.fetch_next_batch()
        ts = batch["timestamp_utc"]
        enriched_map = self.features.ingest_batch(batch["nodes"])

        scored_nodes: List[Dict[str, Any]] = []
        prio_to_num = {
            "OK": 0,
            "MAINTENANCE": 0,
            "P3": 1,
            "P2": 2,
            "P3_WATCH": 1,
            "P1": 3,
        }

        for raw_node in batch["nodes"]:
            nid = raw_node["node_id"]
            ef = enriched_map[nid]

            ml_out = self.ml_engine.score_node(ef)

            alert_out = self.alerts.evaluate(
                node_sample=ef,
                ml_result=ml_out,
                timestamp_utc=ts,
                data_mode=self.connector.mode,
            )

            merged = {
                **ef,
                **ml_out,
                **alert_out,
            }

            scored_nodes.append(merged)

            rack = str(merged.get("rack_id", "rack-1"))

            PROM_FAILURE_PROB.labels(
                node_id=nid,
                rack_id=rack,
            ).set(ml_out["failure_prob_1h"])

            PROM_ANOMALY_SCORE.labels(
                node_id=nid,
                rack_id=rack,
            ).set(ml_out["anomaly_score"])

            PROM_CONSENSUS_VOTES.labels(
                node_id=nid,
            ).set(ml_out["consensus_count"])

            PROM_ALERT_PRIORITY.labels(
                node_id=nid,
            ).set(
                prio_to_num.get(alert_out["alert_priority"], 0)
            )

            PROM_STRAGGLER_SCORE.labels(
                node_id=nid,
            ).set(
                float(merged.get("mpi_straggler_score", 0.0))
            )

            if nid not in self.chart_history:
                self.chart_history[nid] = deque(maxlen=60)

            short_time = ts[11:19] if len(ts) >= 19 else ts

            self.chart_history[nid].append(
                {
                    "time": short_time,
                    "timestamp_utc": ts,
                    "up": merged.get("up", 1),
                    "health_score": merged.get("node_health_score"),
                    "failure_prob_pct": round(
                        ml_out["failure_prob_1h"] * 100.0, 2
                    ),
                    "failure_prob_6h_pct": round(
                        ml_out["failure_prob_6h"] * 100.0, 2
                    ),
                    "anomaly_score_pct": round(
                        ml_out["anomaly_score"] * 100.0, 2
                    ),
                    "cpu": merged.get("cpu_utilization_percent"),
                    "iowait": merged.get("cpu_iowait_percent"),
                    "memory": merged.get("memory_used_percent"),
                    "swap": merged.get("swap_used_percent"),
                    "load1": merged.get("load1"),
                    "load5": merged.get("load5"),
                    "norm_load": merged.get("normalized_load1"),
                    "temp": merged.get("cpu_temperature_celsius"),
                    "root_fs": merged.get("root_used_percent"),
                    "scratch": merged.get("scratch_used_percent"),
                    "disk_read_mb": (
                        round(
                            float(merged["disk_read_bytes_per_second"])
                            / (1024 * 1024),
                            2,
                        )
                        if merged.get("disk_read_bytes_per_second") is not None
                        else None
                    ),
                    "disk_write_mb": (
                        round(
                            float(merged["disk_write_bytes_per_second"])
                            / (1024 * 1024),
                            2,
                        )
                        if merged.get("disk_write_bytes_per_second") is not None
                        else None
                    ),
                    "disk_io": merged.get("disk_io_utilization_percent"),
                    "net_rx_mb": (
                        round(
                            float(merged["network_receive_bytes_per_second"])
                            / (1024 * 1024),
                            2,
                        )
                        if merged.get("network_receive_bytes_per_second") is not None
                        else None
                    ),
                    "net_tx_mb": (
                        round(
                            float(merged["network_transmit_bytes_per_second"])
                            / (1024 * 1024),
                            2,
                        )
                        if merged.get("network_transmit_bytes_per_second") is not None
                        else None
                    ),
                    "net_err": merged.get("network_errors_per_second"),
                    "psi_cpu": merged.get("psi_cpu_percent"),
                    "psi_mem": merged.get("psi_memory_percent"),
                    "psi_io": merged.get("psi_io_percent"),
                    "straggler_pct": round(
                        float(merged.get("mpi_straggler_score", 0.0))
                        * 100.0,
                        1,
                    ),
                }
            )

        # Persist telemetry every second tick.
        if self._tick_counter % 2 == 1:
            try:
                record_telemetry_batch(
                    ts,
                    self.connector.mode,
                    scored_nodes,
                )
            except Exception:
                logger.exception("Could not persist telemetry batch.")

        episodes = list_episodes(limit=60)

        active_eps = [
            e for e in episodes
            if e["status"] in ("active", "acknowledged")
        ]

        # Aggregate cluster-wide KPIs.
        online_nodes = [
            n for n in scored_nodes
            if n.get("up") == 1
        ]

        cpu_vals = [
            float(n["cpu_utilization_percent"])
            for n in online_nodes
            if n.get("cpu_utilization_percent") is not None
        ]

        mem_vals = [
            float(n["memory_used_percent"])
            for n in online_nodes
            if n.get("memory_used_percent") is not None
        ]

        health_vals = [
            float(n["node_health_score"])
            for n in scored_nodes
            if n.get("node_health_score") is not None
        ]

        cluster_cpu = (
            round(statistics.fmean(cpu_vals), 1)
            if cpu_vals else None
        )

        cluster_mem = (
            round(statistics.fmean(mem_vals), 1)
            if mem_vals else None
        )

        avail_ratio = len(online_nodes) / max(len(scored_nodes), 1)

        raw_health = (
            statistics.fmean(health_vals) * avail_ratio
            if health_vals else 0.0
        )

        cluster_health_score = round(
            max(0.0, min(100.0, raw_health)),

        PROM_CLUSTER_HEALTH.set(cluster_health_score)

        if len(online_nodes) == 0:

        peak_risk_pct = round(

        self.cluster_history.append(
            {
                "time": short_time,
                "timestamp_utc": ts,
                "health_score": cluster_health_score,
                "cpu_percent": cluster_cpu,
                "memory_percent": cluster_mem,
                "failure_risk_pct": peak_risk_pct,
                "anomaly_pct": mean_anom_pct,
                "nodes_online": len(online_nodes),
            }
        )

        # Enrich Slurm jobs with blast-radius tags.
        node_risk_lookup = {
            n["node_id"]: n
            for n in scored_nodes
        }

        enriched_jobs = []

        for job in batch.get("slurm_jobs", []):
            jc = dict(job)
            job_nodes = jc.get("nodes", [])

            max_p = max(
                (
                    float(node_risk_lookup[nid]["failure_prob_1h"])
                    for nid in job_nodes
                    if nid in node_risk_lookup

                nid

                and node_risk_lookup[nid]["alert_priority"] in ("P1", "P2")
            ]

            jc["max_node_failure_prob"] = round(max_p, 4)
            jc["at_risk_nodes"] = flagged

            jc["blast_status"] = (
                "CRITICAL_RISK"
                if max_p >= 0.40
                or any(
                    node_risk_lookup.get(nid, {}).get("alert_priority") == "P1"
                    for nid in job_nodes
                )
                else (
                    "WARNING_RISK"
                    if flagged
                    else "HEALTHY"
                )
            )

            enriched_jobs.append(jc)

        settings_cur = get_app_settings()

        # Include the latest cached LSTM V8 state.
        snapshot = {
            "timestamp_utc": ts,
            "mode": self.connector.mode,
            "lstm_v8": self.lstm_v8.get_status(),

            ),
            "prometheus_url": self.connector.prometheus_url,
            "model_status": self.ml_engine.get_status(),
            "ssh_status": get_ssh_public_status(),
          
            "active_fault_injections": self.connector.injected_faults,
            "kpis": {
                "nodes_total": len(scored_nodes),
                "nodes_online": len(online_nodes),
                "nodes_drained": sum(
                    1 for n in scored_nodes
                    if "DRAIN" in str(n.get("slurm_state", ""))
                ),
                "cluster_health_score": cluster_health_score,
                "cluster_health_label": health_label,
                "cluster_cpu_percent": cluster_cpu,
                "cluster_memory_percent": cluster_mem,
                "active_episodes_total": len(active_eps),
                "p1_count": sum(
                    1 for e in active_eps if e["priority"] == "P1"
                ),
                "p2_count": sum(
                    1 for e in active_eps if e["priority"] == "P2"

                "peak_failure_prob_pct": peak_risk_pct,

    async def broadcast(self, payload: Dict[str, Any]) -> None:
        dead: List[WebSocket] = []

        for ws in list(self.clients):
            try:
                await ws.send_json(payload)
            except Exception:
                dead.append(ws)

        for ws in dead:
            self.clients.discard(ws)

@asynccontextmanager
async def lifespan(_app: FastAPI):
    init_db()
    orchestrator.sync_interval_from_settings()

    add_system_log(
        timestamp_utc=datetime.now(timezone.utc).strftime(
            "%Y-%m-%dT%H:%M:%SZ"
        ),
        node_id="head-node",
        source="aiops-daemon",
        severity="INFO",
        event_type="startup",
        message=(
            "Sol Plaatje A Dev Cluster Dashboard v2.0 online. "
            "Real-time telemetry & ML pipeline initialized."
        ),
    )

    # Fetch the initial LSTM V8 state.
    await orchestrator.lstm_v8.refresh_once()

    # Warm up the existing dashboard pipeline.
    warmup_ticks = (
        2
        if orchestrator.connector.mode == "live_prometheus"
        else 14
    )

    for _ in range(warmup_ticks):
        orchestrator.step_once()

    # Start independent LSTM V8 polling.
    await orchestrator.lstm_v8.start()

    stop_event = asyncio.Event()

    async def _loop() -> None:
        while not stop_event.is_set():
            try:
                if not orchestrator.paused:
                    snap = orchestrator.step_once()
                    await orchestrator.broadcast(snap)

            except Exception:
                logger.exception(
                    "Dashboard pipeline iteration failed."
                )

            await asyncio.sleep(
                orchestrator.stream_interval_sec
            )

    task = asyncio.create_task(_loop())

    try:
        yield

    finally:
        stop_event.set()
        task.cancel()

        try:
            await task
        except asyncio.CancelledError:
            pass

        await orchestrator.lstm_v8.stop()

@app.get(
    "/api/monitoring/history",
    dependencies=[Depends(require_authenticated_user)],

def get_monitoring_range_history(
    range_key: str = Query("1h", description="5m | 15m | 1h | 6h | 12h | 24h"),
    node_id: Optional[str] = Query(None),
) -> Dict[str, Any]:
    if range_key not in RANGE_CONFIG:
        range_key = "1h"
    minutes, _step = RANGE_CONFIG[range_key]

    snap = orchestrator.latest_snapshot or orchestrator.step_once()
    nodes = snap.get("nodes", [])
    target_node = next((n for n in nodes if n["node_id"] == node_id), None) if node_id else None
    instance_filter = target_node.get("instance") if target_node else None

    # 1. If connected to real Prometheus, query /api/v1/query_range directly!
    if orchestrator.connector.mode == "live_prometheus" and orchestrator.connector.prometheus_connected:
        prom_series = query_prometheus_range_history(
            prometheus_url=orchestrator.connector.prometheus_url,
            range_key=range_key,
            instance=instance_filter,
        )
        if prom_series is not None:
            return {
                "range_key": range_key,
                "node_id": node_id or "cluster",
                "source": "prometheus_query_range",
                "series": prom_series,
            }

    # 2. Otherwise serve persisted SQLite telemetry_history (or live ring buffer)
    db_rows = query_telemetry_history(range_minutes=minutes, node_id=node_id, limit=720)
    if db_rows:
        # Group by timestamp if cluster-wide, or map directly if single node
        if node_id:
            mapped = [
                {
                    "timestamp_utc": r["timestamp_utc"],
                    "time": r["timestamp_utc"][11:19] if len(r["timestamp_utc"]) >= 19 else r["timestamp_utc"],
                    "up": r["up"],
                    "health_score": r["health_score"],
                    "cpu": r["cpu_percent"],
                    "iowait": r["iowait_percent"],
                    "memory": r["memory_percent"],
                    "swap": r["swap_percent"],
                    "load1": r["load1"],
                    "norm_load": r["normalized_load1"],
                    "scratch": r["scratch_percent"],
                    "disk_read_mb": r["disk_read_mb"],
                    "disk_write_mb": r["disk_write_mb"],
                    "disk_io": r["disk_io_percent"],
                    "net_rx_mb": r["net_rx_mb"],
                    "net_tx_mb": r["net_tx_mb"],
                    "net_err": r["net_err_s"],
                    "temp": r["temp_celsius"],
                    "failure_prob_pct": round(float(r["failure_prob_1h"]) * 100.0, 2) if r["failure_prob_1h"] is not None else None,
                    "anomaly_score_pct": round(float(r["anomaly_score"]) * 100.0, 2) if r["anomaly_score"] is not None else None,
                }
                for r in db_rows
            ]
        else:
            by_ts: Dict[str, List[Dict[str, Any]]] = {}
            for r in db_rows:
                by_ts.setdefault(r["timestamp_utc"], []).append(r)
            mapped = []
            for ts_k in sorted(by_ts.keys()):
                grp = by_ts[ts_k]
                cpus = [float(x["cpu_percent"]) for x in grp if x["cpu_percent"] is not None]
                mems = [float(x["memory_percent"]) for x in grp if x["memory_percent"] is not None]
                hlths = [float(x["health_score"]) for x in grp if x["health_score"] is not None]
                dws = [float(x["disk_write_mb"]) for x in grp if x["disk_write_mb"] is not None]
                nrxs = [float(x["net_rx_mb"]) for x in grp if x["net_rx_mb"] is not None]
                risks = [float(x["failure_prob_1h"]) * 100.0 for x in grp if x["failure_prob_1h"] is not None]
                mapped.append(
                    {
                        "timestamp_utc": ts_k,
                        "time": ts_k[11:19] if len(ts_k) >= 19 else ts_k,
                        "health_score": round(statistics.fmean(hlths), 1) if hlths else None,
                        "cpu": round(statistics.fmean(cpus), 2) if cpus else None,
                        "memory": round(statistics.fmean(mems), 2) if mems else None,
                        "disk_write_mb": round(sum(dws), 2) if dws else None,
                        "net_rx_mb": round(sum(nrxs), 2) if nrxs else None,
                        "failure_prob_pct": round(max(risks), 2) if risks else 0.0,
                    }
                )
        return {
            "range_key": range_key,
            "node_id": node_id or "cluster",
            "source": "recorded_telemetry_db",
            "series": mapped,
        }

    fallback_series = snap.get("history", {}).get(node_id, []) if node_id else snap.get("cluster_history", [])
    return {
        "range_key": range_key,
        "node_id": node_id or "cluster",
        "source": "in_memory_buffer",
        "series": fallback_series,
    }

@app.get(
    "/api/analytics/summary",
    dependencies=[Depends(require_authenticated_user)],
)
def get_performance_analytics_summary(range_key: str = Query("1h")) -> Dict[str, Any]:
    minutes, _ = RANGE_CONFIG.get(range_key, (60, 30))
    snap = orchestrator.latest_snapshot or orchestrator.step_once()
    nodes = snap.get("nodes", [])
    hist_rows = query_telemetry_history(range_minutes=minutes, limit=1500)

    # Compute per-node historical averages & availability %
    by_node: Dict[str, List[Dict[str, Any]]] = {}
    for r in hist_rows:
        by_node.setdefault(r["node_id"], []).append(r)

    node_comparisons = []
    bottlenecks = []
    for n in nodes:
        nid = n["node_id"]
        nrows = by_node.get(nid, [])
        avail_pct = (
            round((sum(1 for x in nrows if x["up"] == 1) / len(nrows)) * 100.0, 2)
            if nrows
            else (100.0 if n.get("up") == 1 else 0.0)
        )
        avg_cpu = _avg_col(nrows, "cpu_percent", n.get("cpu_utilization_percent"))
        avg_mem = _avg_col(nrows, "memory_percent", n.get("memory_used_percent"))
        avg_io = _avg_col(nrows, "iowait_percent", n.get("cpu_iowait_percent"))
        avg_dw = _avg_col(nrows, "disk_write_mb", 0.0)
        avg_nrx = _avg_col(nrows, "net_rx_mb", 0.0)

        node_comparisons.append(
            {
                "node_id": nid,
                "display_name": n.get("display_name") or nid,
                "role": n.get("role", "compute"),
                "availability_pct": avail_pct,
                "avg_cpu": avg_cpu,
                "avg_memory": avg_mem,
                "avg_iowait": avg_io,
                "avg_disk_write_mb": avg_dw,
                "avg_net_rx_mb": avg_nrx,
                "health_score": n.get("node_health_score", 100.0),
            }
        )

        # Bottleneck classification from real observed values
        if (avg_io or 0) >= 10.0:
            bottlenecks.append(
                {
                    "node_id": nid,
                    "type": "Storage / NFS I/O Wait Stall",
                    "severity": "high" if (avg_io or 0) >= 20 else "medium",
                    "detail": f"Mean CPU I/O wait is {avg_io:.1f}% over {range_key}, throttling compute throughput.",
                }
            )
        if (avg_mem or 0) >= 85.0:
            bottlenecks.append(
                {
                    "node_id": nid,
                    "type": "Memory Capacity Pressure",
                    "severity": "high" if (avg_mem or 0) >= 92 else "medium",
                    "detail": f"Mean RAM utilisation at {avg_mem:.1f}%; elevated swap/OOM risk for large MPI ranks.",
                }
            )
        if float(n.get("mpi_straggler_score") or 0) >= 0.45:
            bottlenecks.append(
                {
                    "node_id": nid,
                    "type": "MPI Collective Straggler Imbalance",
                    "severity": "medium",
                    "detail": f"Node trails peer compute median (straggler index {(float(n['mpi_straggler_score'])*100):.0f}%).",
                }
            )

    overall_avail = (
        round(statistics.fmean([c["availability_pct"] for c in node_comparisons]), 2)
        if node_comparisons
        else 100.0
    )

    return {
        "range_key": range_key,
        "overall_availability_pct": overall_avail,
        "node_comparisons": node_comparisons,
        "bottlenecks": bottlenecks,
    }


def _avg_col(rows: List[Dict[str, Any]], key: str, fallback: Any) -> Optional[float]:
    vals = [float(r[key]) for r in rows if r.get(key) is not None]
    if vals:
        return round(statistics.fmean(vals), 2)
    return round(float(fallback), 2) if fallback is not None else None

        "cluster_summary",
        "ai_predictions",
        "alerts_incidents",

            "timestamp_utc": snap.get("timestamp_utc", ""),
            "nodes": snap.get("nodes", []),
            "slurm_jobs": snap.get("slurm_jobs", []),
        },
    }

    node_ids = [
        str(node.get("node_id"))
        for node in snap.get("nodes", [])
        if node.get("node_id")
    ]

    try:
        report_id = save_ai_report(
            report_type=req.report_type,
            title=report_data["title"],
            range_minutes=req.range_minutes,
            data_mode=str(
                snap.get("mode_label")

                or "unknown"

            provider="Google Gemini",
            model=GEMINI_MODEL,
            prompt_version=PROMPT_VERSION,
            node_ids=node_ids,

            report_text="",

        )
    except Exception as exc:
        logger.exception("Could not save pending AI report.")
        raise HTTPException(
            status_code=500,
            detail="Could not create the AI report.",
        ) from exc

    background_tasks.add_task(
        _generate_ai_report_background,

    )

        "status": "pending",
        "title": report_data["title"],

    evidence: Dict[str, Any],

    Generate an AI report outside the original HTTP request.

    try:
        update_ai_report(report_id, "generating")

        result = generate_ai_report(evidence)

        updated = update_ai_report(
            report_id,
            "completed",
            result["report_text"],
        )

        if not updated:
            logger.error(
                "AI report %s disappeared before completion.",
                report_id,
            )

    except Exception:
        logger.exception(
            "Background generation failed for AI report %s.",
            report_id,
        )

        update_ai_report(report_id, "failed")

class ModelEngine:
    """
    Hot-Swappable Model Adapter:
      - Automatically detects and loads .pkl models from backend/models/ if present.
      - Cleanly falls back to a calibrated Multi-Detector Statistical + Trend Ensemble
        when .pkl files have not yet been dropped in, explicitly labeling prediction provenance.
    """

    def __init__(self, models_dir: Path = MODELS_DIR) -> None:
        self.models_dir = models_dir
        self.scaler: Any = None
        self.iforest: Any = None
        self.ocsvm: Any = None
        self.lof: Any = None
        self.kmeans: Any = None
        self.kmeans_threshold: float = 5.07
        self.supervised_model: Any = None
        self.supervised_features: List[str] = []
        self.evaluation_metrics: Optional[Dict[str, Any]] = None
        self.loaded_files: List[str] = []
        self.reload_models()

    def reload_models(self) -> Dict[str, Any]:
        """Scan backend/models/ and load any available .pkl / .json artifacts."""
        self.scaler = None
        self.iforest = None
        self.ocsvm = None
        self.lof = None
        self.kmeans = None
        self.supervised_model = None
        self.supervised_features = []
        self.evaluation_metrics = None
        self.loaded_files = []

        self.models_dir.mkdir(parents=True, exist_ok=True)
        try:
            import joblib  # type: ignore
        except Exception:
            return self.get_status()

        def _try_load(filename: str) -> Any:
            p = self.models_dir / filename
            if p.exists():
                try:
                    obj = joblib.load(p)
                    self.loaded_files.append(filename)
                    return obj
                except Exception:
                    return None
            return None

        self.scaler = _try_load("hpc_feature_scaler.pkl")
        self.iforest = _try_load("isolation_forest_hpc.pkl")
        self.ocsvm = _try_load("one_class_svm_hpc.pkl")
        self.lof = _try_load("lof_hpc.pkl")
        self.kmeans = _try_load("kmeans_hpc.pkl")
        km_thresh = _try_load("kmeans_threshold.pkl")
        if km_thresh is not None:
            try:
                self.kmeans_threshold = float(km_thresh)
            except Exception:
                pass

        self.supervised_model = _try_load("final_xgb_failure_model.pkl") or _try_load(
            "final_rf_failure_model.pkl"
        )
        feat_json = self.models_dir / "rf_feature_columns.json"
        if feat_json.exists():
            try:
                self.supervised_features = json.loads(feat_json.read_text(encoding="utf-8"))
                self.loaded_files.append("rf_feature_columns.json")
            except Exception:
                self.supervised_features = []

        metrics_json = self.models_dir / "model_metrics.json"
        if metrics_json.exists():
            try:
                self.evaluation_metrics = json.loads(metrics_json.read_text(encoding="utf-8"))
                self.loaded_files.append("model_metrics.json")
            except Exception:
                self.evaluation_metrics = None

        return self.get_status()

    def get_status(self) -> Dict[str, Any]:
        has_unsupervised_pkl = any(
            m is not None for m in (self.iforest, self.ocsvm, self.lof, self.kmeans)
        )
        has_supervised_pkl = self.supervised_model is not None
        if has_unsupervised_pkl and has_supervised_pkl:
            mode = "production_pkl_full"
            label = "Trained ML Active (Full .pkl)"
        elif has_unsupervised_pkl or has_supervised_pkl:
            mode = "hybrid_pkl"
            label = "Hybrid ML (.pkl + Baseline)"
        else:
            mode = "placeholder_heuristic"
            label = "Statistical Baseline (Awaiting .pkl)"

        shap_supported = has_supervised_pkl and hasattr(self.supervised_model, "feature_importances_")

        return {
            "engine_mode": mode,
            "engine_label": label,
            "models_directory": str(self.models_dir),
            "loaded_files": self.loaded_files,
            "shap_or_importances_available": shap_supported,
            "evaluation_metrics": self.evaluation_metrics,
            "detectors": {
                "isolation_forest": "pkl" if self.iforest is not None else "statistical",
                "one_class_svm": "pkl" if self.ocsvm is not None else "statistical",
                "lof": "pkl" if self.lof is not None else "statistical",
                "kmeans": "pkl" if self.kmeans is not None else "statistical",
                "failure_predictor_1h": "pkl" if self.supervised_model is not None else "statistical",
            },
        }

    def score_node(self, feats: Dict[str, Any]) -> Dict[str, Any]:
        node_id = str(feats.get("node_id", "unknown"))
        ip = str(feats.get("ip", "127.0.0.1"))
        up = int(feats.get("up", 1) if feats.get("up") is not None else 0)
        planned_maint = int(feats.get("planned_maintenance", 0)) == 1
        has_sup_pkl = self.supervised_model is not None and bool(self.supervised_features)
        pred_source = "trained_pkl_model" if has_sup_pkl else "statistical_baseline"

        if planned_maint:
            pb = PLAYBOOKS["planned_maintenance"]
            return {
                "failure_prob_1h": 0.0,
                "failure_prob_6h": 0.0,
                "failure_prob_24h": 0.0,
                "prediction_confidence": None,
                "prediction_source": pred_source,
                "anomaly_score": 0.0,
                "consensus_count": 0,
                "consensus_votes": {
                    "isolation_forest": False,
                    "one_class_svm": False,
                    "lof": False,
                    "kmeans": False,
                },
                "node_health_score": 100.0,
                "health_classification": "Maintenance",
                "fault_class": "planned_maintenance",
                "fault_label": pb["label"],
                "playbook_cmd": pb["cmd"].format(node=node_id, ip=ip),
                "playbook_advice": pb["advice"],
                "top_features": [],
            }

        if up == 0:
            gt_class = str(feats.get("gt_event_class") or "")
            is_scrape_gap = gt_class == "scrape_gap" or bool(feats.get("simulated_scrape_gap", False))
            fclass = "scrape_gap" if is_scrape_gap else "power_loss"
            pb = PLAYBOOKS[fclass]
            return {
                "failure_prob_1h": 0.08 if is_scrape_gap else 0.99,
                "failure_prob_6h": 0.12 if is_scrape_gap else 0.99,
                "failure_prob_24h": 0.18 if is_scrape_gap else 0.99,
                "prediction_confidence": None,
                "prediction_source": "telemetry_availability_check",
                "anomaly_score": 0.65 if is_scrape_gap else 1.0,
                "consensus_count": 2 if is_scrape_gap else 4,
                "consensus_votes": {
                    "isolation_forest": True,
                    "one_class_svm": True,
                    "lof": not is_scrape_gap,
                    "kmeans": not is_scrape_gap,
                },
                "node_health_score": 45.0 if is_scrape_gap else 0.0,
                "health_classification": "Scrape Gap" if is_scrape_gap else "Offline",
                "fault_class": fclass,
                "fault_label": pb["label"],
                "playbook_cmd": pb["cmd"].format(node=node_id, ip=ip),
                "playbook_advice": pb["advice"],
                "top_features": [
                    {
                        "feature": "up (node_exporter)",
                        "value": 0.0,
                        "unit": "bool",
                        "impact": 0.95,
                        "source_type": "live_observation",
                        "detail": "Exporter unreachable" if is_scrape_gap else "Node offline / unresponsive",
                    }
                ],
            }

        cpu = _num(feats.get("cpu_utilization_percent"), 0.0)
        iowait = _num(feats.get("cpu_iowait_percent"), 0.0)
        mem = _num(feats.get("memory_used_percent"), 0.0)
        mem_slope = _num(feats.get("memory_used_percent__slope12"), 0.0)
        norm_load = _num(feats.get("normalized_load1"), 0.0)
        scratch = _num(feats.get("scratch_used_percent"), _num(feats.get("root_used_percent"), 0.0))
        disk_io = _num(feats.get("disk_io_utilization_percent"), 0.0)
        disk_w = _num(feats.get("disk_write_bytes_per_second"), 0.0)
        net_err = _num(feats.get("network_errors_per_second"), 0.0)
        temp = _num(feats.get("cpu_temperature_celsius"), None)
        temp_slope = _num(feats.get("cpu_temperature_celsius__slope12"), 0.0)
        straggler = _num(feats.get("mpi_straggler_score"), 0.0)
        cluster_busy_mpi = bool(feats.get("cluster_busy_mpi", False))

        # 1. Unsupervised Consensus Scoring
        votes = self._score_unsupervised_detectors(
            feats, cpu, iowait, mem, mem_slope, norm_load, scratch, disk_io, net_err, temp, straggler, cluster_busy_mpi
        )
        consensus_count = sum(1 for v in votes["votes"].values() if v)
        anomaly_score = votes["anomaly_score"]

        # 2. Supervised Failure Risk Scoring + Real Confidence (when supported by .pkl)
        prob_1h, pred_conf, actual_pred_src = self._score_supervised_failure(
            feats, cpu, iowait, mem, mem_slope, norm_load, scratch, disk_io, net_err, temp, temp_slope, straggler
        )
        prob_6h = round(min(0.99, prob_1h * 1.28 + (0.04 if anomaly_score > 0.45 else 0.0)), 4)
        prob_24h = round(min(0.99, prob_6h * 1.22 + (0.03 if mem_slope > 0.2 else 0.0)), 4)

        # 3. Root-Cause Feature Contributions (uses model feature_importances_ weighting if available)
        contributions = self._compute_feature_attributions(
            feats, cpu, iowait, mem, mem_slope, norm_load, scratch, disk_io, disk_w, net_err, temp, temp_slope, straggler
        )
        top_features = contributions[:4]

        # 4. Fault Class Diagnosis & Playbook
        fault_class = _classify_fault(
            prob_1h, anomaly_score, consensus_count, mem, mem_slope, temp, temp_slope, iowait, disk_io, scratch, net_err, norm_load, straggler
        )
        pb = PLAYBOOKS.get(fault_class, PLAYBOOKS["normal"])

        # 5. Deterministic Node Health Score (0-100%)
        health_score, health_cls = _compute_node_health(
            prob_1h=prob_1h,
            anomaly_score=anomaly_score,
            mem=mem,
            iowait=iowait,
            scratch=scratch,
            temp=temp,
            net_err=net_err,
            norm_load=norm_load,
        )

        return {
            "failure_prob_1h": round(prob_1h, 4),
            "failure_prob_6h": prob_6h,
            "failure_prob_24h": prob_24h,
            "prediction_confidence": pred_conf,
            "prediction_source": actual_pred_src,
            "anomaly_score": round(anomaly_score, 4),
            "consensus_count": consensus_count,
            "consensus_votes": votes["votes"],
            "node_health_score": health_score,
            "health_classification": health_cls,
            "fault_class": fault_class,
            "fault_label": pb["label"],
            "playbook_cmd": pb["cmd"].format(node=node_id, ip=ip),
            "playbook_advice": pb["advice"],
            "top_features": top_features,
        }

    def _score_unsupervised_detectors(
        self,
        feats: Dict[str, Any],
        cpu: float,
        iowait: float,
        mem: float,
        mem_slope: float,
        norm_load: float,
        scratch: float,
        disk_io: float,
        net_err: float,
        temp: Optional[float],
        straggler: float,
        cluster_busy_mpi: bool,
    ) -> Dict[str, Any]:
        if self.scaler is not None and any(
            m is not None for m in (self.iforest, self.ocsvm, self.lof, self.kmeans)
        ):
            try:
                import numpy as np  # type: ignore

                row = [[_num(feats.get(col), 0.0) for col in UNSUPERVISED_BASE_FEATURES]]
                x_scaled = self.scaler.transform(row)
                v_if = bool(self.iforest.predict(x_scaled)[0] == -1) if self.iforest is not None else False
                v_svm = bool(self.ocsvm.predict(x_scaled)[0] == -1) if self.ocsvm is not None else False
                v_lof = bool(self.lof.predict(x_scaled)[0] == -1) if self.lof is not None else False
                if self.kmeans is not None:
                    dist = float(np.min(self.kmeans.transform(x_scaled), axis=1)[0])
                    v_km = dist > self.kmeans_threshold
                else:
                    v_km = False

                raw_scores: List[float] = []
                if self.iforest is not None:
                    s_if = -float(self.iforest.decision_function(x_scaled)[0])
                    raw_scores.append(1.0 / (1.0 + math.exp(-6.0 * s_if)))
                if self.ocsvm is not None:
                    s_svm = -float(self.ocsvm.decision_function(x_scaled)[0])
                    raw_scores.append(1.0 / (1.0 + math.exp(-2.0 * s_svm)))

                if (
                    cluster_busy_mpi
                    and straggler < 0.25
                    and iowait < 10.0
                    and mem < 88.0
                    and net_err < 0.5
                    and (temp is None or temp < 82.0)
                ):
                    return {
                        "votes": {
                            "isolation_forest": False,
                            "one_class_svm": False,
                            "lof": False,
                            "kmeans": False,
                        },
                        "anomaly_score": 0.12,
                    }

                votes_dict = {
                    "isolation_forest": v_if,
                    "one_class_svm": v_svm,
                    "lof": v_lof,
                    "kmeans": v_km,
                }
                vote_frac = sum(1 for v in votes_dict.values() if v) / max(
                    sum(1 for m in (self.iforest, self.ocsvm, self.lof, self.kmeans) if m is not None), 1
                )
                agg_score = (sum(raw_scores) / len(raw_scores)) if raw_scores else vote_frac
                return {
                    "votes": votes_dict,
                    "anomaly_score": min(1.0, max(0.0, 0.55 * agg_score + 0.45 * vote_frac)),
                }
            except Exception:
                pass

        temp_excess = max(0.0, (temp - 76.0) / 16.0) if temp is not None else 0.0
        mem_excess = max(0.0, (mem - 82.0) / 15.0) + max(0.0, mem_slope * 0.45)
        io_excess = max(0.0, (iowait - 12.0) / 22.0) + max(0.0, (disk_io - 75.0) / 25.0)
        net_excess = min(1.2, net_err / 2.5)
        load_excess = max(0.0, (norm_load - 1.35) / 0.85)
        scratch_excess = max(0.0, (scratch - 85.0) / 12.0)

        score_if = max(temp_excess, mem_excess, io_excess, net_excess, scratch_excess, load_excess * 0.85)
        score_svm = 0.38 * mem_excess + 0.35 * temp_excess + 0.35 * io_excess + 0.45 * net_excess + 0.30 * load_excess + 0.25 * scratch_excess
        peer_cpu_z = abs(_num(feats.get("cpu_utilization_percent__peer_diff"), 0.0)) / 35.0
        peer_mem_z = abs(_num(feats.get("memory_used_percent__peer_diff"), 0.0)) / 25.0
        score_lof = max(straggler * 1.25, 0.45 * peer_cpu_z + 0.45 * peer_mem_z + 0.35 * io_excess + 0.35 * net_excess)
        if cluster_busy_mpi and straggler < 0.2 and io_excess < 0.1 and temp_excess < 0.1 and mem_excess < 0.15:
            score_lof *= 0.25
        score_km = max(0.5 * score_if + 0.5 * score_svm, straggler * 0.95)

        votes = {
            "isolation_forest": score_if >= 0.48,
            "one_class_svm": score_svm >= 0.46,
            "lof": score_lof >= 0.48,
            "kmeans": score_km >= 0.47,
        }
        raw_combined = 0.32 * min(1.2, score_if) + 0.28 * min(1.2, score_svm) + 0.22 * min(1.2, score_lof) + 0.18 * min(1.2, score_km)
        ambient = 0.04 + 0.03 * min(1.0, cpu / 100.0)
        anomaly_score = min(0.99, ambient + 0.88 * raw_combined)
        return {"votes": votes, "anomaly_score": anomaly_score}

    def _score_supervised_failure(
        self,
        feats: Dict[str, Any],
        cpu: float,
        iowait: float,
        mem: float,
        mem_slope: float,
        norm_load: float,
        scratch: float,
        disk_io: float,
        net_err: float,
        temp: Optional[float],
        temp_slope: float,
        straggler: float,
    ) -> Tuple[float, Optional[float], str]:
        if self.supervised_model is not None and self.supervised_features:
            try:
                import numpy as np  # type: ignore
                import pandas as pd  # type: ignore

                row_dict = {col: _num(feats.get(col), float("nan")) for col in self.supervised_features}
                df_row = pd.DataFrame([row_dict], columns=self.supervised_features)
                prob = float(self.supervised_model.predict_proba(df_row)[0, 1])

                # Compute genuine ensemble tree agreement confidence when supported (RandomForest estimators_)
                confidence: Optional[float] = None
                if hasattr(self.supervised_model, "estimators_"):
                    tree_preds = [
                        float(est.predict_proba(df_row.values)[0, 1])
                        for est in self.supervised_model.estimators_[:50]
                    ]
                    if tree_preds:
                        std_dev = float(np.std(tree_preds))
                        confidence = round(max(0.0, min(1.0, 1.0 - 2.0 * std_dev)), 3)

                return max(0.0, min(1.0, prob)), confidence, "trained_pkl_model"
            except Exception:
                pass

        logit = -5.65
        if mem > 78.0:
            logit += (mem - 78.0) * 0.19
        if mem > 68.0 and mem_slope > 0.35:
            logit += min(2.8, mem_slope * 1.45)
        if temp is not None:
            if temp > 75.0:
                logit += (temp - 75.0) * 0.24
            if temp > 68.0 and temp_slope > 0.25:
                logit += min(2.4, temp_slope * 1.6)
        if iowait > 10.0:
            logit += (iowait - 10.0) * 0.13
        if disk_io > 78.0 and iowait > 8.0:
            logit += (disk_io - 78.0) * 0.11
        if scratch > 86.0:
            logit += (scratch - 86.0) * 0.26
        if net_err > 0.4:
            logit += min(2.5, net_err * 0.85)
        if straggler >= 0.45:
            logit += straggler * 1.65
        if norm_load > 1.5:
            logit += min(1.8, (norm_load - 1.5) * 1.1)

        prob = 1.0 / (1.0 + math.exp(-logit))
        return max(0.0005, min(0.995, prob)), None, "statistical_baseline"

    def _compute_feature_attributions(
        self,
        feats: Dict[str, Any],
        cpu: float,
        iowait: float,
        mem: float,
        mem_slope: float,
        norm_load: float,
        scratch: float,
        disk_io: float,
        disk_w: float,
        net_err: float,
        temp: Optional[float],
        temp_slope: float,
        straggler: float,
    ) -> List[Dict[str, Any]]:
        has_importances = (
            self.supervised_model is not None
            and hasattr(self.supervised_model, "feature_importances_")
            and bool(self.supervised_features)
        )
        attr_type = "model_feature_importance" if has_importances else "deviation_attribution"

        candidates: List[Dict[str, Any]] = []
        mem_impact = max(0.0, (mem - 70.0) / 30.0) + max(0.0, mem_slope * 0.28)
        candidates.append(
            {
                "feature": "memory_used_percent",
                "value": round(mem, 1),
                "unit": "%",
                "impact": round(min(1.0, mem_impact), 3),
                "source_type": attr_type,
                "detail": f"Slope {mem_slope:+.2f}%/step | Peer Î” {_num(feats.get('memory_used_percent__peer_diff'), 0.0):+.1f}%",
            }
        )

        if temp is not None:
            temp_impact = max(0.0, (temp - 68.0) / 25.0) + max(0.0, temp_slope * 0.30)
            candidates.append(
                {
                    "feature": "cpu_temperature_celsius",
                    "value": round(temp, 1),
                    "unit": "Â°C",
                    "impact": round(min(1.0, temp_impact), 3),
                    "source_type": attr_type,
                    "detail": f"Slope {temp_slope:+.2f}Â°C/step | Peer Î” {_num(feats.get('cpu_temperature_celsius__peer_diff'), 0.0):+.1f}Â°C",
                }
            )

        io_impact = max(0.0, (iowait - 5.0) / 28.0) + max(0.0, (disk_io - 65.0) / 40.0)
        candidates.append(
            {
                "feature": "cpu_iowait_percent",
                "value": round(iowait, 2),
                "unit": "%",
                "impact": round(min(1.0, io_impact), 3),
                "source_type": attr_type,
                "detail": f"Disk active {disk_io:.1f}% | Write {disk_w / (1024 * 1024):.1f} MB/s",
            }
        )

        scratch_impact = max(0.0, (scratch - 75.0) / 22.0)
        candidates.append(
            {
                "feature": "scratch_used_percent",
                "value": round(scratch, 1),
                "unit": "%",
                "impact": round(min(1.0, scratch_impact), 3),
                "source_type": attr_type,
                "detail": "Filesystem pool critical limit 90%",
            }
        )

        net_impact = min(1.0, net_err / 3.0)
        candidates.append(
            {
                "feature": "network_errors_per_second",
                "value": round(net_err, 3),
                "unit": "err/s",
                "impact": round(net_impact, 3),
                "source_type": attr_type,
                "detail": "10GbE cluster interconnect packet health",
            }
        )

        load_impact = max(0.0, (norm_load - 1.1) / 1.1)
        candidates.append(
            {
                "feature": "normalized_load1",
                "value": round(norm_load, 2),
                "unit": "x cores",
                "impact": round(min(1.0, max(load_impact, straggler * 0.9)), 3),
                "source_type": attr_type,
                "detail": f"CPU {cpu:.1f}% | MPI Straggler Index {straggler:.2f}",
            }
        )

        candidates.sort(key=lambda c: c["impact"], reverse=True)
        return candidates


def _compute_node_health(
    prob_1h: float,
    anomaly_score: float,
    mem: float,
    iowait: float,
    scratch: float,
    temp: Optional[float],
    net_err: float,
    norm_load: float,
) -> Tuple[float, str]:
    penalty = 0.0
    penalty += prob_1h * 55.0
    penalty += max(0.0, anomaly_score - 0.15) * 24.0
    if mem > 85.0:
        penalty += (mem - 85.0) * 1.1
    if iowait > 10.0:
        penalty += (iowait - 10.0) * 0.6
    if scratch > 85.0:
        penalty += (scratch - 85.0) * 1.4
    if temp is not None and temp > 76.0:
        penalty += (temp - 76.0) * 1.3
    if net_err > 0.2:
        penalty += min(15.0, net_err * 3.0)
    if norm_load > 1.4:
        penalty += min(12.0, (norm_load - 1.4) * 8.0)

    score = round(max(5.0, min(100.0, 100.0 - penalty)), 1)
    if score >= 92.0:
        cls = "Excellent"
    elif score >= 78.0:
        cls = "Healthy"
    elif score >= 55.0:
        cls = "Warning"
    else:
        cls = "Critical"
    return score, cls


def _classify_fault(
    prob_1h: float,
    anomaly_score: float,
    consensus_count: int,
    mem: float,
    mem_slope: float,
    temp: Optional[float],
    temp_slope: float,
    iowait: float,
    disk_io: float,
    scratch: float,
    net_err: float,
    norm_load: float,
    straggler: float,
) -> str:
    if prob_1h < 0.015 and consensus_count < 2 and anomaly_score < 0.42 and straggler < 0.45 and scratch < 88.0:
        return "normal"

    scores = {
        "thermal_shutdown": ((temp - 75.0) / 12.0 + max(0.0, temp_slope * 0.4)) if temp is not None else -1.0,
        "memory_exhaustion": (mem - 78.0) / 12.0 + max(0.0, mem_slope * 0.5),
        "disk_failure": ((iowait - 14.0) / 15.0 + (disk_io - 75.0) / 20.0) if disk_io > 65.0 else (iowait - 20.0) / 15.0,
        "io_congestion": max((scratch - 85.0) / 8.0, (iowait - 9.0) / 14.0),
        "network_degradation": net_err / 1.2,
        "mpi_straggler": straggler * 1.6 if straggler >= 0.45 else -0.5,
        "cpu_contention": (norm_load - 1.35) / 0.5,
    }
    best_cls = max(scores.items(), key=lambda kv: kv[1])
    return best_cls[0] if best_cls[1] > 0.1 else ("mpi_straggler" if straggler >= 0.45 else "io_congestion")


def _num(val: Any, default: Any = 0.0) -> Any:
    if val is None or val == "":
        return default
    try:
        f = float(val)
        return default if math.isnan(f) else f
    except Exception:
        return default

import React, { useEffect, useMemo, useState } from 'react';
import { api, createLiveSocket } from './api.js';

import Sidebar from './components/Sidebar.jsx';

export default function App() {
  // Authentication state
  const [authStatus, setAuthStatus] = useState('checking');
  const [user, setUser] = useState(null);

  // Cluster state
  const [snapshot, setSnapshot] = useState(null);
  const [wsConnected, setWsConnected] = useState(false);

  // Navigation state
  const [activePage, setActivePage] = useState('overview');

  const [sidebarCollapsed, setSidebarCollapsed] = useState(() => {
    try {
      return localStorage.getItem('spu_sidebar_collapsed') === '1';
    } catch {
      return false;
    }
  });

  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
  const [selectedNodeId, setSelectedNodeId] = useState('head-node');

  // Dashboard interaction state
  const [faultDrawerOpen, setFaultDrawerOpen] = useState(false);
  const [busyAction, setBusyAction] = useState(false);
  const [toast, setToast] = useState(null);
  const [confirmState, setConfirmState] = useState({ open: false });

  const showToast = (msg) => {
    setToast(msg);
    setTimeout(() => setToast(null), 3600);
  };

  // ---------------------------------------------------------
  // 1. CHECK EXISTING AUTHENTICATION SESSION
  // ---------------------------------------------------------

  useEffect(() => {
    let mounted = true;

    const checkSession = async () => {
      try {
        const result = await api.getMe();

        if (!mounted) return;

        if (result?.authenticated && result?.user) {
          setUser(result.user);
          setAuthStatus('authenticated');
        } else {
          setUser(null);
          setAuthStatus('unauthenticated');
        }
      } catch {
        if (mounted) {
          setUser(null);
          setAuthStatus('unauthenticated');
        }
      }
    };

    checkSession();

    return () => {
      mounted = false;
    };
  }, []);

  // ---------------------------------------------------------
  // 2. START CLUSTER TELEMETRY AFTER AUTHENTICATION
  // ---------------------------------------------------------

  useEffect(() => {
    if (!user || authStatus !== 'authenticated') return;

    let mounted = true;

    // Load initial cluster state
    api
      .getState()
      .then((data) => {
        if (!mounted) return;

        setSnapshot(data);

        if (data?.nodes?.length > 0) {
          setSelectedNodeId(data.nodes[0].node_id);
        }
      })
      .catch((err) => {
        if (mounted) {
          showToast(`Cluster connection failed: ${err.message}`);
        }
      });

    // Establish live telemetry connection
    const cleanup = createLiveSocket(
      (liveData) => {
        if (mounted) {
          setSnapshot(liveData);
        }
      },
      (status) => {
        if (mounted) {
          setWsConnected(status);
        }
      }
    );

    return () => {
      mounted = false;

      if (typeof cleanup === 'function') {
        cleanup();
      }

      setWsConnected(false);
    };
  }, [user, authStatus]);

  // ---------------------------------------------------------
  // 3. KEEP SELECTED NODE VALID
  // ---------------------------------------------------------

  useEffect(() => {
    if (!snapshot?.nodes?.length) return;

    const exists = snapshot.nodes.some(
      (node) => node.node_id === selectedNodeId
    );

    if (!exists) {
      setSelectedNodeId(snapshot.nodes[0].node_id);
    }
  }, [snapshot?.mode, snapshot?.nodes, selectedNodeId]);

  // Compute nodes exclude management nodes
  const computeNodes = useMemo(
    () => (snapshot?.nodes || []).filter(
      (node) => node.role !== 'management'
    ),
    [snapshot?.nodes]
  );

  // ---------------------------------------------------------
  // 4. AUTHENTICATION HANDLERS
  // ---------------------------------------------------------

  const handleAuthenticated = (authenticatedUser) => {
    setSnapshot(null);
    setUser(authenticatedUser);
    setAuthStatus('authenticated');
    setActivePage('overview');
  };

  const handleLogout = async () => {
    try {
      await api.logout();
    } catch (err) {
      showToast(`Logout error: ${err.message}`);
    } finally {
      setUser(null);
      setSnapshot(null);
      setWsConnected(false);
      setFaultDrawerOpen(false);
      setConfirmState({ open: false });
      setActivePage('overview');
      setMobileMenuOpen(false);
      setAuthStatus('unauthenticated');
    }
  };

  // ---------------------------------------------------------
  // 5. SIDEBAR HANDLERS
  // ---------------------------------------------------------

  const toggleSidebarCollapse = () => {
    setSidebarCollapsed((prev) => {
      const next = !prev;

      try {
        localStorage.setItem(
          'spu_sidebar_collapsed',
          next ? '1' : '0'
        );
      } catch {
        // Ignore unavailable localStorage
      }

      return next;
    });
  };

  // ---------------------------------------------------------
  // 6. CLUSTER ACTION HANDLERS
  // ---------------------------------------------------------

  const handleManualRefresh = async () => {
    setBusyAction(true);

    try {
      const data = await api.refreshState();
      setSnapshot(data);
      showToast('Cluster telemetry refreshed');
    } catch (err) {
      showToast(`Refresh failed: ${err.message}`);
    } finally {
      setBusyAction(false);
    }
  };

  const handleSwitchMode = async (
    mode,
    promUrl = null,
    presetId = null
  ) => {
    setBusyAction(true);

    try {
      const data = await api.switchMode(mode, promUrl, presetId);
      setSnapshot(data);

      showToast(
        `Switched data pipeline to ${data.mode_label || mode}`
      );
    } catch (err) {
      showToast(`Mode switch failed: ${err.message}`);
    } finally {
      setBusyAction(false);
    }
  };

  const handleReloadModels = async () => {
    setBusyAction(true);

    try {
      const status = await api.reloadModels();

      showToast(
        `Model Registry Reloaded: ${status.engine_label} (${status.loaded_files.length} artifacts)`
      );
    } catch (err) {
      showToast(`Model reload failed: ${err.message}`);
    } finally {
      setBusyAction(false);
    }
  };

  const handleTogglePause = async () => {
    try {
      await api.togglePause();
    } catch (err) {
      showToast(`Stream toggle error: ${err.message}`);
    }
  };

  // ---------------------------------------------------------
  // 7. SAFE CONFIRMED ADMINISTRATIVE ACTIONS
  // ---------------------------------------------------------

  const promptDrainNode = (nodeObj = null) => {
    if (nodeObj) {
      setConfirmState({
        open: true,
        title: `Drain ${nodeObj.display_name || nodeObj.node_id} via Slurm`,
        description: `This marks ${nodeObj.node_id} (${nodeObj.ip}) as DRAIN in Slurm so no new compute jobs are scheduled onto it while running jobs finish safely.`,
        confirmLabel: 'Confirm Slurm Drain',
        variant: 'warning',
        requireReason: true,
        defaultReason: 'Proactive maintenance / AI risk mitigation',

        onConfirm: async ({ reason }) => {
          setConfirmState({ open: false });

          try {
            const res = await api.drainNode(nodeObj.node_id, reason);

            showToast(
              `Drained ${nodeObj.node_id} (${res.executed_mode}) â€” Protected ${res.jobs_protected} job(s)`
            );
          } catch (err) {
            showToast(`Drain error: ${err.message}`);
          }
        },
      });
    } else {
      setConfirmState({
        open: true,
        title: 'Drain Compute Node via Slurm',
        description:
          'Select a compute node to gracefully drain from the active Slurm scheduling pool.',
        confirmLabel: 'Confirm Slurm Drain',
        variant: 'warning',
        requireReason: true,
        defaultReason: 'Operator initiated node drain',
        nodePickerOptions: computeNodes,

        onConfirm: async ({ reason, nodeId }) => {
          setConfirmState({ open: false });

          if (!nodeId) return;

          try {
            const res = await api.drainNode(nodeId, reason);

            showToast(
              `Drained ${nodeId} (${res.executed_mode})`
            );
          } catch (err) {
            showToast(`Drain error: ${err.message}`);
          }
        },
      });
    }
  };

  const promptResumeNode = (nodeObj) => {
    setConfirmState({
      open: true,
      title: `Resume ${nodeObj.display_name || nodeObj.node_id} in Slurm`,
      description: `Return ${nodeObj.node_id} (${nodeObj.ip}) to active Slurm job scheduling?`,

        }
      },

      confirmLabel: 'Cancel Slurm Job',
      variant: 'danger',
      requireReason: true,
      defaultReason: 'Cancelled by cluster administrator',

      onConfirm: async ({ reason }) => {
        setConfirmState({ open: false });

        try {
          await api.cancelJob(job.job_id, reason);
          showToast(`Cancelled Slurm job #${job.job_id}`);
        } catch (err) {
          showToast(`Job cancel failed: ${err.message}`);
        }
      },
    });
  };

  // ---------------------------------------------------------
  // 8. ALERT AND FAULT INJECTION HANDLERS
  // ---------------------------------------------------------

  const handleAlertAction = async (episodeId, action) => {
    try {
      await api.alertAction(episodeId, action);
      showToast(`Incident #${episodeId}: ${action} applied`);
    } catch (err) {
      showToast(`Alert update failed: ${err.message}`);
    }
  };

  const handleInjectFault = async (nodeId, faultType) => {
    try {
      await api.injectFault(nodeId, faultType);

      setSelectedNodeId(nodeId);
      showToast(`Injected '${faultType}' on ${nodeId}`);
    } catch (err) {
      showToast(`Fault inject failed: ${err.message}`);
    }
  };

  const handleClearFaults = async (nodeId = null) => {
    try {
      await api.clearFaults(nodeId);
      showToast('Cleared injected faults');
    } catch (err) {
      showToast(`Clear failed: ${err.message}`);
    }
  };

  // ---------------------------------------------------------
  // 9. NODE NAVIGATION HANDLERS
  // ---------------------------------------------------------

  const navigateToNodeMonitoring = (nodeId) => {
    setSelectedNodeId(nodeId);
    setActivePage('monitoring');
  };

  const navigateToNodeLogs = (nodeId) => {
    setSelectedNodeId(nodeId);
    setActivePage('logs');
  };

    setSelectedNodeId(nodeId);
    setActivePage('terminal');

  // ---------------------------------------------------------

  return (
    <>
      {/* Session verification screen */}
      {authStatus === 'checking' && (
        <div className="flex min-h-screen items-center justify-center bg-[var(--bg-app)]">
          <div className="flex flex-col items-center gap-3">
            <div className="h-8 w-8 animate-spin rounded-full border-2 border-[var(--accent-cyan)] border-t-transparent" />

            <p className="text-sm spu-text-secondary">
              Verifying your dashboard session...
            </p>
          </div>

import { LockKeyhole, Mail, ShieldCheck, Loader2 } from 'lucide-react';

export default function AuthPage({ onAuthenticated }) {

  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [code, setCode] = useState('');
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const [message, setMessage] = useState('');

  const handleLogin = async (event) => {
    event.preventDefault();
    setError('');
    setMessage('');
    setLoading(true);

    try {
      const result = await api.login(email.trim(), password);
      setStage('otp');
      setMessage(result?.message || 'A verification code has been sent to your email.');
    } catch (err) {
      setError(err.message || 'Unable to sign in.');
    } finally {
      setLoading(false);
    }
  };

  const handleVerifyOtp = async (event) => {
    event.preventDefault();
    setError('');
    setLoading(true);

    try {
      await api.verifyOtp(email.trim(), code.trim());
      const result = await api.getMe();

      if (!result?.authenticated || !result?.user) {
        throw new Error('Your session could not be verified. Please sign in again.');
      }

      onAuthenticated(result.user);
    } catch (err) {
      setError(err.message || 'OTP verification failed.');
    } finally {
      setLoading(false);
    }
  };

  const handleBack = () => {
    setStage('login');
    setCode('');
    setError('');
    setMessage('');
  };

  return (
    <div className="flex min-h-screen items-center justify-center bg-[var(--bg-app)] px-4 py-10">
      <div className="w-full max-w-md">
        <div className="mb-8 text-center">

            <ShieldCheck size={32} className="text-cyan-400" />
          </div>

            HPC AIOps Dashboard
          </h1>
          <p className="mt-2 text-sm spu-text-secondary">
            Sol Plaatje A Dev Cluster
          </p>
        </div>

        <div className="rounded-2xl border border-[var(--border-color)] bg-[var(--bg-card)] p-6 shadow-xl sm:p-8">
          {stage === 'login' ? (
            <>
              <div className="mb-6">
                <h2 className="text-xl font-semibold spu-text-primary">
                  Sign in
                </h2>
                <p className="mt-1 text-sm spu-text-secondary">
                  Sign in using your registered account.
                </p>
              </div>

              <form onSubmit={handleLogin} className="space-y-5">
                <div>
                  <label className="mb-2 block text-sm font-medium spu-text-primary">
                    Email address
                  </label>
                  <div className="relative">
                    <Mail className="absolute left-3 top-3.5 text-gray-400" size={17} />
                    <input
                      type="email"
                      required
                      autoComplete="username"
                      value={email}
                      onChange={(e) => setEmail(e.target.value)}
                      placeholder="you@example.com"
                      className="w-full rounded-xl border border-[var(--border-color)] bg-[var(--bg-app)] py-3 pl-10 pr-3 text-sm spu-text-primary outline-none focus:border-cyan-500"
                    />

                <div>
                  <label className="mb-2 block text-sm font-medium spu-text-primary">

          ) : (

              <div className="mb-6">
                <h2 className="text-xl font-semibold spu-text-primary">

                node_sample=ef,
                ml_result=ml_out,
                timestamp_utc=ts,
                data_mode=self.connector.mode,
            )

            merged = {
                **ef,
                **ml_out,
                **alert_out,
            }
            scored_nodes.append(merged)

            rack = str(merged.get("rack_id", "rack-1"))
            PROM_FAILURE_PROB.labels(node_id=nid, rack_id=rack).set(ml_out["failure_prob_1h"])
            PROM_ANOMALY_SCORE.labels(node_id=nid, rack_id=rack).set(ml_out["anomaly_score"])
            PROM_CONSENSUS_VOTES.labels(node_id=nid).set(ml_out["consensus_count"])

    warmup_ticks = 2 if orchestrator.connector.mode == "live_prometheus" else 14
    for _ in range(warmup_ticks):
        orchestrator.step_once()

    stop_event = asyncio.Event()

    async def _loop() -> None:
        while not stop_event.is_set():
            try:
                if not orchestrator.paused:
                    snap = orchestrator.step_once()
                    await orchestrator.broadcast(snap)
            except Exception:
                pass
            await asyncio.sleep(orchestrator.stream_interval_sec)

    task = asyncio.create_task(_loop())
    try:
        yield
    finally:
        stop_event.set()
        task.cancel()

        )
        if prom_series is not None:
            return {
                "range_key": range_key,
                "node_id": node_id or "cluster",
                "source": "prometheus_query_range",
                "series": prom_series,
            }

    # 2. Otherwise serve persisted SQLite telemetry_history (or live ring buffer)
    db_rows = query_telemetry_history(range_minutes=minutes, node_id=node_id, limit=720)
    if db_rows:
        # Group by timestamp if cluster-wide, or map directly if single node
        if node_id:
            mapped = [
                {
                    "timestamp_utc": r["timestamp_utc"],
                    "time": r["timestamp_utc"][11:19] if len(r["timestamp_utc"]) >= 19 else r["timestamp_utc"],
                    "up": r["up"],
                    "health_score": r["health_score"],
                    "cpu": r["cpu_percent"],
                    "iowait": r["iowait_percent"],
                    "memory": r["memory_percent"],
                    "swap": r["swap_percent"],
                    "load1": r["load1"],
                    "norm_load": r["normalized_load1"],
                    "scratch": r["scratch_percent"],
                    "disk_read_mb": r["disk_read_mb"],
                    "disk_write_mb": r["disk_write_mb"],
                    "disk_io": r["disk_io_percent"],
                    "net_rx_mb": r["net_rx_mb"],
                    "net_tx_mb": r["net_tx_mb"],
                    "net_err": r["net_err_s"],
                    "temp": r["temp_celsius"],
                    "failure_prob_pct": round(float(r["failure_prob_1h"]) * 100.0, 2) if r["failure_prob_1h"] is not None else None,
                    "anomaly_score_pct": round(float(r["anomaly_score"]) * 100.0, 2) if r["anomaly_score"] is not None else None,
                }
                for r in db_rows
            ]
        else:
            by_ts: Dict[str, List[Dict[str, Any]]] = {}
            for r in db_rows:
                by_ts.setdefault(r["timestamp_utc"], []).append(r)
            mapped = []
            for ts_k in sorted(by_ts.keys()):
                grp = by_ts[ts_k]
                cpus = [float(x["cpu_percent"]) for x in grp if x["cpu_percent"] is not None]
                mems = [float(x["memory_percent"]) for x in grp if x["memory_percent"] is not None]
                hlths = [float(x["health_score"]) for x in grp if x["health_score"] is not None]
                dws = [float(x["disk_write_mb"]) for x in grp if x["disk_write_mb"] is not None]
                nrxs = [float(x["net_rx_mb"]) for x in grp if x["net_rx_mb"] is not None]
                risks = [float(x["failure_prob_1h"]) * 100.0 for x in grp if x["failure_prob_1h"] is not None]
                mapped.append(
                    {
                        "timestamp_utc": ts_k,
                        "time": ts_k[11:19] if len(ts_k) >= 19 else ts_k,
                        "health_score": round(statistics.fmean(hlths), 1) if hlths else None,
                        "cpu": round(statistics.fmean(cpus), 2) if cpus else None,
                        "memory": round(statistics.fmean(mems), 2) if mems else None,
                        "disk_write_mb": round(sum(dws), 2) if dws else None,
                        "net_rx_mb": round(sum(nrxs), 2) if nrxs else None,
                        "failure_prob_pct": round(max(risks), 2) if risks else 0.0,
                    }
                )
        return {
            "range_key": range_key,
            "node_id": node_id or "cluster",
            "source": "recorded_telemetry_db",
            "series": mapped,
        }

    fallback_series = snap.get("history", {}).get(node_id, []) if node_id else snap.get("cluster_history", [])
    return {
        "range_key": range_key,
        "node_id": node_id or "cluster",
        "source": "in_memory_buffer",
        "series": fallback_series,
    }

@app.get("/api/analytics/summary")
def get_performance_analytics_summary(range_key: str = Query("1h")) -> Dict[str, Any]:
    minutes, _ = RANGE_CONFIG.get(range_key, (60, 30))
    snap = orchestrator.latest_snapshot or orchestrator.step_once()
    nodes = snap.get("nodes", [])
    hist_rows = query_telemetry_history(range_minutes=minutes, limit=1500)

    # Compute per-node historical averages & availability %
    by_node: Dict[str, List[Dict[str, Any]]] = {}
    for r in hist_rows:
        by_node.setdefault(r["node_id"], []).append(r)

    node_comparisons = []
    bottlenecks = []
    for n in nodes:
        nid = n["node_id"]
        nrows = by_node.get(nid, [])
        avail_pct = (
            round((sum(1 for x in nrows if x["up"] == 1) / len(nrows)) * 100.0, 2)
            if nrows
            else (100.0 if n.get("up") == 1 else 0.0)
        )
        avg_cpu = _avg_col(nrows, "cpu_percent", n.get("cpu_utilization_percent"))
        avg_mem = _avg_col(nrows, "memory_percent", n.get("memory_used_percent"))
        avg_io = _avg_col(nrows, "iowait_percent", n.get("cpu_iowait_percent"))
        avg_dw = _avg_col(nrows, "disk_write_mb", 0.0)
        avg_nrx = _avg_col(nrows, "net_rx_mb", 0.0)

        node_comparisons.append(
            {
                "node_id": nid,
                "display_name": n.get("display_name") or nid,
                "role": n.get("role", "compute"),
                "availability_pct": avail_pct,
                "avg_cpu": avg_cpu,
                "avg_memory": avg_mem,
                "avg_iowait": avg_io,
                "avg_disk_write_mb": avg_dw,
                "avg_net_rx_mb": avg_nrx,
                "health_score": n.get("node_health_score", 100.0),
            }
        )

        # Bottleneck classification from real observed values
        if (avg_io or 0) >= 10.0:
            bottlenecks.append(
                {
                    "node_id": nid,
                    "type": "Storage / NFS I/O Wait Stall",
                    "severity": "high" if (avg_io or 0) >= 20 else "medium",
                    "detail": f"Mean CPU I/O wait is {avg_io:.1f}% over {range_key}, throttling compute throughput.",
                }
            )
        if (avg_mem or 0) >= 85.0:
            bottlenecks.append(
                {
                    "node_id": nid,
                    "type": "Memory Capacity Pressure",
                    "severity": "high" if (avg_mem or 0) >= 92 else "medium",
                    "detail": f"Mean RAM utilisation at {avg_mem:.1f}%; elevated swap/OOM risk for large MPI ranks.",
                }
            )
        if float(n.get("mpi_straggler_score") or 0) >= 0.45:
            bottlenecks.append(
                {
                    "node_id": nid,
                    "type": "MPI Collective Straggler Imbalance",
                    "severity": "medium",
                    "detail": f"Node trails peer compute median (straggler index {(float(n['mpi_straggler_score'])*100):.0f}%).",
                }
            )

    overall_avail = (
        round(statistics.fmean([c["availability_pct"] for c in node_comparisons]), 2)
        if node_comparisons
        else 100.0
    )

    return {
        "range_key": range_key,
        "overall_availability_pct": overall_avail,
        "node_comparisons": node_comparisons,
        "bottlenecks": bottlenecks,
    }


def _avg_col(rows: List[Dict[str, Any]], key: str, fallback: Any) -> Optional[float]:
    vals = [float(r[key]) for r in rows if r.get(key) is not None]
    if vals:
        return round(statistics.fmean(vals), 2)
    return round(float(fallback), 2) if fallback is not None else None

@app.post(
    "/api/reports/ai/generate",
    dependencies=[Depends(require_admin_role)],
)
def generate_ai_report_endpoint(
    req: AIReportGenerateRequest,
    background_tasks: BackgroundTasks,
) -> Dict[str, Any]:

    allowed_types = {
        "cluster_summary",
        "ai_predictions",
        "alerts_incidents",
        "slurm_workload",
        "performance_history",
    }

    if req.report_type not in allowed_types:
        raise HTTPException(
            status_code=400,
            detail="Unsupported report type.",
        )

    snap = orchestrator.latest_snapshot or orchestrator.step_once()

    report_data = build_report_data(
        report_type=req.report_type,
        snapshot=snap,
        range_minutes=req.range_minutes,
    )

    evidence = {
        "report_data": report_data,
        "cluster_context": {
            "mode": snap.get("mode", ""),
            "mode_label": snap.get("mode_label", ""),
            "timestamp_utc": snap.get("timestamp_utc", ""),
            "nodes": snap.get("nodes", []),
            "slurm_jobs": snap.get("slurm_jobs", []),
        },
    }

    node_ids = [
        str(node.get("node_id"))
        for node in snap.get("nodes", [])
        if node.get("node_id")
    ]

    try:
        report_id = save_ai_report(
            report_type=req.report_type,
            title=report_data["title"],
            range_minutes=req.range_minutes,
            data_mode=str(
                snap.get("mode_label")
                or snap.get("mode")
                or "unknown"
            ),
            provider="Google Gemini",
            model=GEMINI_MODEL,
            prompt_version=PROMPT_VERSION,
            node_ids=node_ids,
            evidence=evidence,
            report_text="",
            status="pending",
        )
    except Exception as exc:
        logger.exception("Could not save pending AI report.")
        raise HTTPException(
            status_code=500,
            detail="Could not create the AI report.",
        ) from exc

    background_tasks.add_task(
        _generate_ai_report_background,
        report_id,
        evidence,
    )

    return {
        "report_id": report_id,
        "status": "pending",
        "title": report_data["title"],
        "message": "AI report generation has been queued.",
    }

def _generate_ai_report_background(
    report_id: int,
    evidence: Dict[str, Any],
) -> None:
    """
    Generate an AI report outside the original HTTP request.
    """

    try:
        update_ai_report(report_id, "generating")

        result = generate_ai_report(evidence)

        updated = update_ai_report(
            report_id,
            "completed",
            result["report_text"],
        )

        if not updated:
            logger.error(
                "AI report %s disappeared before completion.",
                report_id,
            )

    except Exception:
        logger.exception(
            "Background generation failed for AI report %s.",
            report_id,
        )

        update_ai_report(report_id, "failed")

async def update_settings_endpoint(req: SettingsUpdateRequest) -> Dict[str, Any]:
    updates = {k: v for k, v in req.model_dump().items() if v is not None}
    saved = update_app_settings(updates)
    orchestrator.sync_interval_from_settings()
    orchestrator.connector.sync_from_settings()
    snap = orchestrator.step_once()
    await orchestrator.broadcast(snap)
    return {"saved": True, "settings": saved}

@app.websocket("/ws/live")
async def websocket_live_stream(ws: WebSocket) -> None:
    await ws.accept()
    orchestrator.clients.add(ws)
    try:
        if orchestrator.latest_snapshot:
            await ws.send_json(orchestrator.latest_snapshot)
        while True:
            await ws.receive_text()
    except WebSocketDisconnect:
        orchestrator.clients.discard(ws)
    except Exception:
        orchestrator.clients.discard(ws)

        if full_path.startswith("api/"):
            raise HTTPException(status_code=404, detail="API endpoint not found")
        candidate = FRONTEND_DIST_DIR / full_path
        if full_path and candidate.is_file():
